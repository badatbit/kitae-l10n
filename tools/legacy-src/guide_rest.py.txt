def cells(t):
    return sum(0.5 if len(c.encode("cp932", "replace")) == 1 else 1 for c in t)


def main():
    docs = {}
    for f in ("guide", "guide2", "guides"):
        with io.open(os.path.join(GUIDE, f + ".json"), encoding="utf-8") as fh:
            docs[f] = json.load(fh)
    by = {(f, e["window"], e["line"]): e
          for f, d in docs.items() for e in d["entries"]}

    bad = []
    for k, ko in KO.items():
        if k not in by:
            bad.append(f"{k}: 그런 자리가 없다")
        elif cells(ko) > MAX:
            bad.append(f"{k}: {cells(ko)}칸 > {MAX}  {ko}")
        elif (by[k]["text"].get("ko") or "").strip():
            bad.append(f"{k}: 이미 번역돼 있다")
    if bad:
        print(f"검사 실패 {len(bad)}건")
        for b in bad:
            print("  " + b)
        return 1

    fixed = 0
    for d in docs.values():
        for e in d["entries"]:
            ko = e["text"].get("ko") or ""
            if ko.startswith(TYPO[0]):
                e["text"]["ko"] = TYPO[1] + ko[len(TYPO[0]):]
                fixed += 1

    for k, ko in KO.items():
        by[k]["text"]["ko"] = ko
    for k, why in ASK.items():
        if k in by:
            by[k]["ask"] = [{"why": why, "state": "open"}]

    tot = done = 0
    for f, d in docs.items():
        with io.open(os.path.join(GUIDE, f + ".json"), "w", encoding="utf-8") as fh:
            fh.write(json.dumps(d, ensure_ascii=False, indent=2) + "\n")
        tot += len(d["entries"])
        done += sum(1 for e in d["entries"] if (e["text"].get("ko") or "").strip())
    print(f"검사 통과 — {len(KO)}줄 채우고 `이그` 오타 {fixed}줄 고침. "
          f"관광지 {done}/{tot}줄 ({done*100//tot}%), 검토 요청 {len(ASK)}건")
    return 0


if __name__ == "__main__":
    sys.exit(main())
