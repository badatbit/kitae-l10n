def cells(s):
    return sum(0.5 if len(c.encode("cp932", "replace")) == 1 else 1
               for c in MARKUP.sub("", s or ""))


doc = json.load(io.open(DST, encoding="utf-8"))
by = {}
for e in doc["entries"]:
    by.setdefault(e["window"], {})[e["line"]] = e

bad = []
for oid, (qlines, choices) in KO.items():
    rec = by.get(oid)
    if rec is None:
        bad.append(f"[{oid}] 그런 문제가 없다")
        continue
    ja = rec[0]["text"]["ja"]
    tail = len(ja) - len(ja.rstrip("\n"))
    ko = "\n".join(qlines) + "\n" * tail
    if ja.count("\n") != ko.count("\n"):
        bad.append(f"[{oid}] 줄 수가 다르다 {ja.count(chr(10))} → {ko.count(chr(10))}")
    for l in qlines:
        if cells(l) > Q_MAX:
            bad.append(f"[{oid}] 문제 {cells(l)}칸 > {Q_MAX}  {l}")
    for i, c in enumerate(choices):
        if cells(c) > C_MAX:
            bad.append(f"[{oid}] 선택지{i+1} {cells(c)}칸 > {C_MAX}  {c}")
    if len(set(choices)) != 3:
        bad.append(f"[{oid}] 선택지가 겹친다 {choices}")

if bad:
    print(f"검사 실패 {len(bad)}건")
    for b in bad:
        print("  " + b)
    raise SystemExit(1)

for oid, (qlines, choices) in KO.items():
    rec = by[oid]
    ja = rec[0]["text"]["ja"]
    tail = len(ja) - len(ja.rstrip("\n"))
    rec[0]["text"]["ko"] = "\n".join(qlines) + "\n" * tail
    for i, c in enumerate(choices):
        rec[i + 1]["text"]["ko"] = c

    by[oid][0]["ask"] = [{"why": why, "state": "open"}]
